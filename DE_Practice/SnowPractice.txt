-- ============================================================
-- STEP 1: Create the project database and layered schemas
-- ============================================================
USE ROLE SYSADMIN;

CREATE DATABASE IF NOT EXISTS RETAILMART_DW
  COMMENT = 'RetailMart Global Data Warehouse - Practice Project';

-- Bronze / Silver / Gold / Reporting layers
CREATE SCHEMA IF NOT EXISTS RETAILMART_DW.RAW
  COMMENT = 'Bronze: raw untransformed source data';
CREATE SCHEMA IF NOT EXISTS RETAILMART_DW.STAGING
  COMMENT = 'Silver: cleansed and validated data';
CREATE SCHEMA IF NOT EXISTS RETAILMART_DW.ANALYTICS
  COMMENT = 'Gold: star schema fact and dimension tables';
CREATE SCHEMA IF NOT EXISTS RETAILMART_DW.REPORTING
  COMMENT = 'Aggregated views for BI tools';
CREATE SCHEMA IF NOT EXISTS RETAILMART_DW.CONTROL
  COMMENT = 'ETL control tables: watermarks, audit logs';


-- ============================================================
-- STEP 2: Create purpose-specific warehouses
-- ============================================================
USE ROLE SYSADMIN;

-- Ingestion warehouse (used for COPY INTO, Snowpipe)
CREATE WAREHOUSE IF NOT EXISTS INGEST_WH
  WAREHOUSE_SIZE = 'SMALL'
  AUTO_SUSPEND = 15
  AUTO_RESUME = TRUE
  COMMENT = 'Used for data loading and ingestion';


desc warehouse INGEST_WH;

-- Transformation warehouse (heavy ETL, dbt, Streams+Tasks)
CREATE WAREHOUSE IF NOT EXISTS TRANSFORM_WH
  WAREHOUSE_SIZE = 'MEDIUM'
  AUTO_SUSPEND = 15
  AUTO_RESUME = TRUE
  COMMENT = 'Heavy transformation and pipeline processing';

-- Analytics / BI warehouse (dashboards, ad-hoc queries)
CREATE WAREHOUSE IF NOT EXISTS ANALYTICS_WH
  WAREHOUSE_SIZE = 'SMALL'
  AUTO_SUSPEND = 15
  AUTO_RESUME = TRUE
  INITIALLY_SUSPENDED = TRUE
  COMMENT = 'Analyst queries and BI tool connections';

-- High-concurrency warehouse for multi-user dashboards
CREATE WAREHOUSE IF NOT EXISTS BI_WH
  WAREHOUSE_SIZE = 'SMALL'
  AUTO_SUSPEND = 15
  AUTO_RESUME = TRUE
  COMMENT = 'Multi-cluster for concurrent BI users';



-- ============================================================
-- STEP 3: Create custom roles for each team
-- ============================================================
USE ROLE SECURITYADMIN;

-- Create functional roles
CREATE ROLE IF NOT EXISTS DE_ENGINEER_ROLE   COMMENT = 'Data Engineers: full ETL access';
CREATE ROLE IF NOT EXISTS DA_ANALYST_ROLE    COMMENT = 'Analysts: read analytics + reporting';
CREATE ROLE IF NOT EXISTS BI_ROLE            COMMENT = 'BI tools: read reporting schemas only';
CREATE ROLE IF NOT EXISTS PII_ADMIN_ROLE     COMMENT = 'Can see unmasked PII columns';

USE ROLE DE_ENGINEER_ROLE;
USE DATABASE RETAILMART_DW;
USE SCHEMA RAW;

-- Raw customers (source: CRM system)
CREATE OR REPLACE TABLE raw_customers (
  load_id         NUMBER AUTOINCREMENT PRIMARY KEY,
  customer_id     VARCHAR(20),
  full_name       VARCHAR(200),
  email           VARCHAR(255),
  phone           VARCHAR(30),
  address         VARCHAR(500),
  city            VARCHAR(100),
  state           VARCHAR(50),
  country         VARCHAR(50),
  postal_code     VARCHAR(20),
  date_of_birth   VARCHAR(20),       -- raw: may be malformed
  gender          VARCHAR(10),
  loyalty_tier    VARCHAR(20),
  signup_date     VARCHAR(30),       -- raw: string
  is_active       VARCHAR(5),        -- raw: 'Y'/'N'/'true'/'false'
  raw_json        VARIANT,           -- full source JSON for reference
  _load_ts        TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  _source_file    VARCHAR(500)
);

-- Raw products (source: Product catalog API)
CREATE OR REPLACE TABLE raw_products (
  load_id         NUMBER AUTOINCREMENT PRIMARY KEY,
  product_id      VARCHAR(20),
  sku             VARCHAR(50),
  product_name    VARCHAR(500),
  category        VARCHAR(100),
  sub_category    VARCHAR(100),
  brand           VARCHAR(100),
  unit_price      VARCHAR(20),       -- raw: may include '$' symbol
  cost_price      VARCHAR(20),
  weight_kg       VARCHAR(20),
  stock_qty       VARCHAR(20),
  supplier_id     VARCHAR(20),
  country_of_origin VARCHAR(50),
  tags            VARIANT,           -- JSON array of tags
  attributes      VARIANT,           -- JSON object of specs
  _load_ts        TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  _source_file    VARCHAR(500)
);

-- Raw orders (source: Transactional DB)
CREATE OR REPLACE TABLE raw_orders (
  load_id         NUMBER AUTOINCREMENT PRIMARY KEY,
  order_id        VARCHAR(30),
  customer_id     VARCHAR(20),
  order_date      VARCHAR(30),
  status          VARCHAR(30),
  channel         VARCHAR(30),       -- 'web','mobile','store','marketplace'
  region          VARCHAR(50),
  currency        VARCHAR(5),
  subtotal        VARCHAR(20),
  discount_amt    VARCHAR(20),
  tax_amt         VARCHAR(20),
  shipping_amt    VARCHAR(20),
  total_amt       VARCHAR(20),
  payment_method  VARCHAR(30),
  promo_code      VARCHAR(30),
  notes           VARCHAR(1000),
  line_items      VARIANT,           -- JSON array of order items
  _load_ts        TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  _source_file    VARCHAR(500)
);

-- Raw clickstream events (source: Web analytics)
CREATE OR REPLACE TABLE raw_events (
  load_id         NUMBER AUTOINCREMENT PRIMARY KEY,
  event_id        VARCHAR(40),
  session_id      VARCHAR(40),
  user_id         VARCHAR(20),
  event_type      VARCHAR(50),  -- page_view, add_to_cart, purchase, search
  event_payload   VARIANT,      -- full JSON event body
  event_ts        TIMESTAMP,
  _load_ts        TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);



-- Dimension: Date (pre-populated, never changes)
CREATE OR REPLACE TABLE dim_date (
  date_key        NUMBER PRIMARY KEY,   -- YYYYMMDD format
  full_date       DATE,
  year            NUMBER(4),
  quarter         NUMBER(1),
  month           NUMBER(2),
  month_name      VARCHAR(10),
  week_of_year    NUMBER(2),
  day_of_week     NUMBER(1),
  day_name        VARCHAR(10),
  day_of_month    NUMBER(2),
  is_weekend      BOOLEAN,
  is_holiday      BOOLEAN DEFAULT FALSE,
  fiscal_year     NUMBER(4),
  fiscal_quarter  NUMBER(1)
);

-- Dimension: Customer (SCD Type 2)
CREATE OR REPLACE TABLE dim_customer (
  customer_sk     NUMBER AUTOINCREMENT PRIMARY KEY,  -- surrogate key
  customer_id     VARCHAR(20)  NOT NULL,             -- business key
  full_name       VARCHAR(200),
  email           VARCHAR(255),
  phone           VARCHAR(30),
  city            VARCHAR(100),
  state           VARCHAR(50),
  country         VARCHAR(50),
  postal_code     VARCHAR(20),
  date_of_birth   DATE,
  gender          VARCHAR(10),
  loyalty_tier    VARCHAR(20),  -- BRONZE/SILVER/GOLD/PLATINUM
  age_group       VARCHAR(20),  -- 18-24, 25-34, 35-44, 45-54, 55+
  signup_date     DATE,
  is_active       BOOLEAN,
  -- SCD Type 2 columns
  effective_from  DATE,
  effective_to    DATE,
  is_current      BOOLEAN DEFAULT TRUE,
  -- Audit
  created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Dimension: Product
CREATE OR REPLACE TABLE dim_product (
  product_sk      NUMBER AUTOINCREMENT PRIMARY KEY,
  product_id      VARCHAR(20)  NOT NULL,
  sku             VARCHAR(50),
  product_name    VARCHAR(500),
  category        VARCHAR(100),
  sub_category    VARCHAR(100),
  brand           VARCHAR(100),
  unit_price      NUMBER(12,2),
  cost_price      NUMBER(12,2),
  gross_margin_pct NUMBER(5,2),
  weight_kg       NUMBER(8,3),
  supplier_id     VARCHAR(20),
  country_of_origin VARCHAR(50),
  is_active       BOOLEAN DEFAULT TRUE,
  created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Fact: Orders (grain = one order line item)
CREATE OR REPLACE TABLE fact_orders (
  order_sk        NUMBER AUTOINCREMENT PRIMARY KEY,
  order_id        VARCHAR(30)  NOT NULL,
  order_line_num  NUMBER(3)    NOT NULL,
  -- Foreign keys to dimensions
  date_key        NUMBER       REFERENCES dim_date(date_key),
  customer_sk     NUMBER       REFERENCES dim_customer(customer_sk),
  product_sk      NUMBER       REFERENCES dim_product(product_sk),
  -- Degenerate dimensions
  status          VARCHAR(30),
  channel         VARCHAR(30),
  region          VARCHAR(50),
  payment_method  VARCHAR(30),
  promo_code      VARCHAR(30),
  -- Measures
  quantity        NUMBER(6),
  unit_price      NUMBER(12,2),
  line_subtotal   NUMBER(14,2),
  discount_amt    NUMBER(12,2),
  tax_amt         NUMBER(12,2),
  shipping_amt    NUMBER(12,2),
  line_total      NUMBER(14,2),
  cost_total      NUMBER(14,2),
  gross_profit    NUMBER(14,2),
  -- Audit
  loaded_at       TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Control table: ETL watermarks
USE SCHEMA CONTROL;
CREATE OR REPLACE TABLE etl_watermark (
  table_name      VARCHAR(200) PRIMARY KEY,
  last_loaded_ts  TIMESTAMP,

  rows_processed  NUMBER,

CREATE OR REPLACE TABLE etl_audit_log (

  run_id          VARCHAR(100),
  start_time      TIMESTAMP,

  rows_inserted   NUMBER DEFAULT 0,
  rows_updated    NUMBER DEFAULT 0,
  rows_deleted    NUMBER DEFAULT 0,
  error_message   VARCHAR(2000),
  created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Generate date dimension for 2022-01-01 to 2025-12-31
INSERT INTO dim_date
WITH date_spine AS (
  SELECT DATEADD(DAY, SEQ4(), '2022-01-01'::DATE) AS dt
  FROM TABLE(GENERATOR(ROWCOUNT => 1461))  -- 4 years
)
SELECT
  TO_NUMBER(TO_CHAR(dt, 'YYYYMMDD'))        AS date_key,
  dt                                         AS full_date,
  YEAR(dt)                                   AS year,
  QUARTER(dt)                                AS quarter,
  MONTH(dt)                                  AS month,
  TO_CHAR(dt, 'MMMM')                        AS month_name,
  WEEKOFYEAR(dt)                             AS week_of_year,
  DAYOFWEEK(dt)                              AS day_of_week,
  DAYNAME(dt)                                AS day_name,
  DAYOFMONTH(dt)                             AS day_of_month,
  CASE WHEN DAYOFWEEK(dt) IN (1,7) THEN TRUE ELSE FALSE END AS is_weekend,
  FALSE                                      AS is_holiday,
  CASE WHEN MONTH(dt) >= 4 THEN YEAR(dt) ELSE YEAR(dt)-1 END AS fiscal_year,
  CASE
    WHEN MONTH(dt) IN (4,5,6)   THEN 1
    WHEN MONTH(dt) IN (7,8,9)   THEN 2
    WHEN MONTH(dt) IN (10,11,12) THEN 3
    ELSE 4
  END                                        AS fiscal_quarter
FROM date_spine;

-- Mark major US holidays

5.2 Customer Dimension Load Task;
USE SCHEMA RETAILMART_DW.ANALYTICS;

-- Task: transform and load customers from stream into dim_customer
CREATE OR REPLACE TASK load_dim_customer
  WAREHOUSE = TRANSFORM_WH
  SCHEDULE = '10 MINUTE'
  WHEN SYSTEM$STREAM_HAS_DATA('RETAILMART_DW.RAW.CUSTOMERS_STREAM')
AS
BEGIN
  -- Step 1: Merge new/updated customers into dimension
  MERGE INTO RETAILMART_DW.ANALYTICS.dim_customer AS target
  USING (
    SELECT DISTINCT
      customer_id,
      full_name,
      email,
      phone,
      city, state, country, postal_code,
      TRY_TO_DATE(date_of_birth, 'YYYY-MM-DD') AS date_of_birth,
      UPPER(gender) AS gender,
      UPPER(loyalty_tier) AS loyalty_tier,
      CASE
        WHEN DATEDIFF('year', TRY_TO_DATE(date_of_birth,'YYYY-MM-DD'), CURRENT_DATE) < 25 THEN '18-24'
        WHEN DATEDIFF('year', TRY_TO_DATE(date_of_birth,'YYYY-MM-DD'), CURRENT_DATE) < 35 THEN '25-34'
        WHEN DATEDIFF('year', TRY_TO_DATE(date_of_birth,'YYYY-MM-DD'), CURRENT_DATE) < 45 THEN '35-44'
        WHEN DATEDIFF('year', TRY_TO_DATE(date_of_birth,'YYYY-MM-DD'), CURRENT_DATE) < 55 THEN '45-54'
        ELSE '55+'
      END AS age_group,
      TRY_TO_DATE(signup_date, 'YYYY-MM-DD') AS signup_date,
      CASE WHEN UPPER(is_active) IN ('Y','YES','TRUE','1') THEN TRUE ELSE FALSE END AS is_active,
      METADATA$ACTION
    FROM RETAILMART_DW.RAW.customers_stream
    WHERE METADATA$ACTION = 'INSERT'
  ) AS source
  ON target.customer_id = source.customer_id AND target.is_current = TRUE

  -- Update: close old record if key attributes changed
  WHEN MATCHED AND (
    target.email <> source.email OR
    target.loyalty_tier <> source.loyalty_tier OR
    target.city <> source.city
  ) THEN UPDATE SET
    target.is_current = FALSE,
    target.effective_to = CURRENT_DATE - 1,
    target.updated_at = CURRENT_TIMESTAMP

  -- Insert: new customer or SCD Type 2 new version
  WHEN NOT MATCHED THEN INSERT (
    customer_id, full_name, email, phone,
    city, state, country, postal_code,
    date_of_birth, gender, loyalty_tier, age_group,
    signup_date, is_active, effective_from, is_current
  ) VALUES (
    source.customer_id, source.full_name, source.email, source.phone,
    source.city, source.state, source.country, source.postal_code,
    source.date_of_birth, source.gender, source.loyalty_tier, source.age_group,
    source.signup_date, source.is_active, CURRENT_DATE, TRUE
  );

  -- Step 2: Log the run
  INSERT INTO RETAILMART_DW.CONTROL.etl_audit_log
    (pipeline_name, run_id, start_time, end_time, status, rows_inserted)
  VALUES ('load_dim_customer', CURRENT_STATEMENT(), CURRENT_TIMESTAMP,
          CURRENT_TIMESTAMP, 'SUCCESS', 0);
END;

-- Resume task (tasks start in suspended state)
ALTER TASK load_dim_customer RESUME;

CREATE OR REPLACE TASK load_fact_orders
  WAREHOUSE = TRANSFORM_WH
  SCHEDULE = '15 MINUTE'
  WHEN SYSTEM$STREAM_HAS_DATA('RETAILMART_DW.RAW.ORDERS_STREAM')
AS
  INSERT INTO RETAILMART_DW.ANALYTICS.fact_orders (
    order_id, order_line_num, date_key, customer_sk, product_sk,
    status, channel, region, payment_method, promo_code,
    quantity, unit_price, line_subtotal, discount_amt,
    tax_amt, shipping_amt, line_total, cost_total, gross_profit
  )
  SELECT
    o.order_id,
    f.index + 1                                              AS order_line_num,
    TO_NUMBER(TO_CHAR(TO_DATE(o.order_date,'YYYY-MM-DD'),'YYYYMMDD')) AS date_key,
    c.customer_sk,
    p.product_sk,
    o.status,
    o.channel,
    o.region,
    o.payment_method,
    o.promo_code,
    f.value:qty::NUMBER                                      AS quantity,
    f.value:unit_price::NUMBER(12,2)                         AS unit_price,
    f.value:qty::NUMBER * f.value:unit_price::NUMBER(12,2)   AS line_subtotal,
    -- Prorate discount proportionally across lines
    ROUND(o.discount_amt::FLOAT *
      (f.value:qty::NUMBER * f.value:unit_price::NUMBER(12,2)) /
      NULLIF(o.subtotal::FLOAT, 0), 2)                       AS discount_amt,
    ROUND(o.tax_amt::FLOAT *
      (f.value:qty::NUMBER * f.value:unit_price::NUMBER(12,2)) /
      NULLIF(o.subtotal::FLOAT, 0), 2)                       AS tax_amt,
    ROUND(o.shipping_amt::FLOAT *
      (f.value:qty::NUMBER * f.value:unit_price::NUMBER(12,2)) /
      NULLIF(o.subtotal::FLOAT, 0), 2)                       AS shipping_amt,
    f.value:qty::NUMBER * f.value:unit_price::NUMBER(12,2)   AS line_total,
    f.value:qty::NUMBER * pr.cost_price                      AS cost_total,
    (f.value:qty * f.value:unit_price::NUMBER(12,2)) -
    (f.value:qty * pr.cost_price)                            AS gross_profit
  FROM RETAILMART_DW.RAW.orders_stream o,
  LATERAL FLATTEN(input => o.line_items) f
  LEFT JOIN RETAILMART_DW.ANALYTICS.dim_customer c
    ON o.customer_id = c.customer_id AND c.is_current = TRUE
  LEFT JOIN RETAILMART_DW.ANALYTICS.dim_product p
    ON f.value:product_id::STRING = p.product_id
  LEFT JOIN RETAILMART_DW.RAW.raw_products pr
    ON f.value:product_id::STRING = pr.product_id
  WHERE o.METADATA$ACTION = 'INSERT';

ALTER TASK load_fact_orders RESUME;
