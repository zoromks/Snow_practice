creating storage integration and stage from azure to snowflake:


CREATE OR REPLACE STORAGE INTEGRATION azure_int
TYPE = EXTERNAL_STAGE
STORAGE_PROVIDER = 'AZURE'
ENABLED = TRUE
AZURE_TENANT_ID = 'bb0cf065-408e-478a-833a-8cdd5b835175'
STORAGE_ALLOWED_LOCATIONS = (
'azure://blobmks.blob.core.windows.net/blob1/'
);


CREATE OR REPLACE FILE FORMAT csv_ff
TYPE = CSV
FIELD_DELIMITER = ','
SKIP_HEADER = 1
FIELD_OPTIONALLY_ENCLOSED_BY = '"';



CREATE OR REPLACE STAGE adf_stage
URL='azure://blobmks.blob.core.windows.net/blob1/'
STORAGE_INTEGRATION = azure_int
FILE_FORMAT = csv_ff;


list @adf_stage;
